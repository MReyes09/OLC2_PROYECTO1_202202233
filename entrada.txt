func main() {
    // Enteros
    a := 10
    b := 4
    c := 3

    // Flotantes
    x := 2.5
    y := 1.5
    z := 4.0

    // Operaciones entre enteros
    suma := a + b              // 10 + 4 = 14
    resta := a - c             // 10 - 3 = 7
    multi := b * c             // 4 * 3 = 12
    div := a / b               // 10 / 4 = 2
    mod := a % c               // 10 % 3 = 1

    // Operaciones entre floats
    sumaF := x + y             // 2.5 + 1.5 = 4.0
    restaF := z - y            // 4.0 - 1.5 = 2.5
    multiF := x * z            // 2.5 * 4.0 = 10.0
    divF := z / x              // 4.0 / 2.5 = 1.6

    // Mezclas
    mezcla := float64(a) + x * float64(b) - y  // 10 + (2.5 * 4) - 1.5 = 18.5

    // Prints
    println(suma)
    println(resta)
    println(multi)
    println(div)
    println(mod)

    println(sumaF)
    println(restaF)
    println(multiF)
    println(divF)

    println(mezcla)
}
