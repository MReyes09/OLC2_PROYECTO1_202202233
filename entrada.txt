fn main() {
    // Declaración e inicialización de variables
    mut var1 := 10
    mut var2 := 0.0
    // Correcto: Operación += con valores del mismo tipo
    var1 += 10 // var1 tendrá el valor de 20
    println('var1: $var1')
    // Error: No se puede asignar un float a un int (comentado)
    // var1 += 10.0
    // Correcto: Operación += con valores float
    var2 += 10 // var2 tendrá el valor de 10.0
    println('var2: $var2')
    // Correcto: Operación += con valores del mismo tipo (float)
    var2 += 10.0 // var2 tendrá el valor de 20.0
    println('var2: $var2')
    // Declaración de una cadena de texto
    mut str := 'cad'
    // Correcto: Concatenación de strings con +=
    str += 'cad' // str tendrá el valor de "cadcad"
    println('str: $str')
    // Error: Operación inválida string + int (comentado)
    // str += 10
}
