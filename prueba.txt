// Probando aritmeticos
func main(){
  println("Suma simple enteros")
  println(10+15)
  println("")

  println("Suma simple float")
  println(1.65 + 0.15)
  println("")

  println("Suma entre strings")
  println("Hola" + " ¿Como estas?")
  println("")

  println("Suma entre tipos distintos")
  println("15 + 1.5 =",15 + 1.5)
  println("6.99 + 1 =", 6.99 + 1)
  println("")

  println("Resta simple enteros")
  println(10-15)
  println("")

  println("Resta simple float")
  println(1.65 - 0.15)
  println("")

  println("Suma entre tipos distintos")
  println("15 - 1.5 =",15 - 1.5)
  println("")
}

// * | / | %
func main(){
  // Multiplicación
    println("5 * 2 =", 5 * 2)

    // División
    println("10 / 2 =", 10 / 2)

    // Módulo
    println("9 % 4 =", 9 % 4)

    // Float multiplicación
    println("5.5 * 2.0 =", 5.5 * 2.0)

    // Float división
    println("10.0 / 4.0 =", 10.0 / 4.0)

    // Mezcla int + float
    println("5 * 2.5 =", 5 * 2.5)
}
